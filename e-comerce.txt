<script>
const products = [
  { id:1, name:'Wireless Headphones', price:2499, cat:'Audio', icon:'fa-headphones' },
  { id:2, name:'Bluetooth Speaker', price:1899, cat:'Audio', icon:'fa-volume-high' },
  { id:3, name:'Gaming Headset', price:2199, cat:'Audio', icon:'fa-headset' },
  { id:4, name:'Smartphone Earbuds', price:1599, cat:'Audio', icon:'fa-headphones-simple' },
  { id:5, name:'Smart Watch', price:4999, cat:'Wearables', icon:'fa-clock' },
  { id:6, name:'Fitness Band', price:1799, cat:'Wearables', icon:'fa-heart-pulse' },
  { id:7, name:'Power Bank 20000mAh', price:1299, cat:'Accessories', icon:'fa-battery-full' },
  { id:8, name:'Wireless Mouse', price:649, cat:'Accessories', icon:'fa-computer-mouse' },
  { id:9, name:'USB-C Hub', price:1099, cat:'Accessories', icon:'fa-plug' },
  { id:10, name:'Laptop Stand', price:899, cat:'Accessories', icon:'fa-laptop' },
  { id:11, name:'Portable SSD 1TB', price:3499, cat:'Storage', icon:'fa-hard-drive' },
  { id:12, name:'MicroSD Card 256GB', price:1199, cat:'Storage', icon:'fa-memory' },
];
 
function fmt(n){
  return n.toLocaleString('en-US') + ' Br';
}
 
const cats = ['All', ...new Set(products.map(p=>p.cat))];
let cart = [];
let activeCat = 'All';
 
function renderFilters(){
  const el = document.getElementById('filters');
  el.innerHTML = cats.map(c =>
    `<button class="filter-btn ${c===activeCat?'active':''}" onclick="setFilter('${c}')">${c}</button>`
  ).join('');
}
 
function setFilter(cat){
  activeCat = cat;
  renderFilters();
  renderGrid();
}
 
function renderGrid(){
  const list = activeCat==='All' ? products : products.filter(p=>p.cat===activeCat);
  const grid = document.getElementById('productGrid');
  grid.innerHTML = list.map(p => `
    <div class="card">
      <div class="card-visual"><i class="fas ${p.icon}"></i></div>
      <div class="card-body">
        <span class="card-cat">${p.cat}</span>
        <span class="card-name">${p.name}</span>
        <div class="card-footer">
          <span class="price">${fmt(p.price)}</span>
          <button class="add-btn" id="btn-${p.id}" onclick="addToCart(${p.id})">Add to Cart</button>
        </div>
      </div>
    </div>
  `).join('');
}
 
function addToCart(id){
  const product = products.find(p=>p.id===id);
  const existing = cart.find(i=>i.id===id);
  if(existing){ existing.qty++; } else { cart.push({...product, qty:1}); }
  updateCartUI();
  showToast(`${product.name} added to cart`);
  const btn = document.getElementById(`btn-${id}`);
  btn.classList.add('added'); btn.textContent = 'Added ✓';
  setTimeout(()=>{ btn.classList.remove('added'); btn.textContent='Add to Cart'; }, 1200);
}
 
function changeQty(id, delta){
  const item = cart.find(i=>i.id===id);
  if(!item) return;
  item.qty += delta;
  if(item.qty <= 0) cart = cart.filter(i=>i.id!==id);
  updateCartUI();
}
 
function removeItem(id){
  cart = cart.filter(i=>i.id!==id);
  updateCartUI();
}
 
function cartTotal(){
  return cart.reduce((sum,i)=> sum + i.price*i.qty, 0);
}
 
function updateCartUI(){
  const count = cart.reduce((s,i)=>s+i.qty,0);
  document.getElementById('cartBadge').textContent = count;
 
  const wrap = document.getElementById('cartItems');
  if(cart.length===0){
    wrap.innerHTML = `<div class="cart-empty"><i class="fas fa-cart-shopping"></i>Your cart is empty</div>`;
  } else {
    wrap.innerHTML = cart.map(i => `
      <div class="cart-item">
        <div class="ci-visual"><i class="fas ${i.icon}"></i></div>
        <div class="ci-info">
          <div class="ci-name">${i.name}</div>
          <div class="ci-price">${fmt(i.price)}</div>
          <div class="qty-ctrl">
            <button onclick="changeQty(${i.id},-1)">−</button>
            <span>${i.qty}</span>
            <button onclick="changeQty(${i.id},1)">+</button>
            <button class="remove-btn" onclick="removeItem(${i.id})" style="margin-left:8px;">Remove</button>
          </div>
        </div>
      </div>
    `).join('');
  }
  const total = cartTotal();
  document.getElementById('subtotal').textContent = fmt(total);
  document.getElementById('checkoutBtn').disabled = cart.length===0;
}
 
function openCart(){ document.getElementById('drawer').classList.add('open'); document.getElementById('overlay').classList.add('open'); }
function closeCart(){ document.getElementById('drawer').classList.remove('open'); document.getElementById('overlay').classList.remove('open'); }
 
function openCheckout(){
  if(cart.length===0) return;
  document.getElementById('modalTotal').textContent = fmt(cartTotal());
  showStep('checkoutForm');
  document.getElementById('checkoutOverlay').classList.add('open');
}
function closeCheckout(){
  document.getElementById('checkoutOverlay').classList.remove('open');
  cart = [];
  updateCartUI();
  closeCart();
  pin = '';
}
 
function showStep(id){
  ['checkoutForm','tbPhoneStep','tbPinStep','tbProcessing','successView'].forEach(s=>{
    document.getElementById(s).style.display = (s===id) ? 'block' : 'none';
  });
}
 
function goToPayment(){
  const name = document.getElementById('cName').value.trim();
  const address = document.getElementById('cAddress').value.trim();
  if(!name || !address){ showToast('Please fill in your name and address'); return; }
  document.getElementById('tbAmount1').textContent = fmt(cartTotal());
  showStep('tbPhoneStep');
}
 
function goToPin(){
  const phone = document.getElementById('tbPhone').value.trim();
  if(!/^0?9\d{8}$/.test(phone)){ showToast('Enter a valid telebirr number (09xxxxxxxx)'); return; }
  document.getElementById('tbPhoneDisplay').textContent = 'Line: ' + phone;
  pin = '';
  updatePinDots();
  showStep('tbPinStep');
}
 
let pin = '';
function updatePinDots(){
  document.querySelectorAll('.pin-dot').forEach((d,i)=> d.classList.toggle('filled', i < pin.length));
}
function pressKey(k){
  if(pin.length >= 4) return;
  pin += k;
  updatePinDots();
  if(pin.length === 4){
    showStep('tbProcessing');
    setTimeout(()=>{
      document.getElementById('successMsg').textContent =
        `${fmt(cartTotal())} was paid successfully via telebirr. Your order is on its way!`;
      showStep('successView');
    }, 1600);
  }
}
function backspacePin(){ pin = pin.slice(0,-1); updatePinDots(); }
function clearPin(){ pin = ''; updatePinDots(); }
 
function showToast(msg){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  clearTimeout(window._toastTimer);
  window._toastTimer = setTimeout(()=> t.classList.remove('show'), 1800);
}
 
renderFilters();
renderGrid();
updateCartUI();
